\begin{frame}[plain]{Motivation: Can you trust Neural Networks for Medical Image Processing?}

	\note{
		\begin{enumerate}
			\item Deep Neural Networks have become an essential part of Medical Image Processing
			\item Tasks: Image Segmentation, Registration, Reconstruction

\begin{frame}[plain]{Perturbation Tests: Insertion and Deletion \cite{petsiuk_rise_2018}}
	\note{
		\begin{itemize}
			\item perturbation test proposed in the literature to evaluate fidelity
			\item start with trained model and AD sample
			\item forward pass through model to get prediction
			\item also compute attribution map
			\item use attribution map to perturb input image
			\item mask voxels in descending order of their relevance, starting with most relevant voxels
			\item hippocampus masked, expect predicted probabilit to decrease
			\item repeat until input image is completely masked
		\end{itemize}
	}

\begin{frame}[plain]{}
	\note{
		\begin{itemize}
			\item null image popular choice for attribution baseline and perturbation target
			\item attribution maps fail to decrease predicted probability for Alzheimer’s disease
			\item hypothesis: null image is unnatural image of maximum atrophy
			\item Model just associates weaker signal with atrophy and Alzheimer’s disease
			\item compared to null image, every other image has more signals
			\item also explains some weird observations regarding sign of the relevance
			\item so, what should actually be used as baseline?
		\end{itemize}
	}
